        // Storage Controller
        class StorageManager {
            static load() {
                try {
                    const data = localStorage.getItem('vyra_state_v1');
                    if (!data) return DEFAULT_STATE;
                    const parsed = JSON.parse(data);
                    return { ...DEFAULT_STATE, ...parsed };
                } catch (e) {
                    console.error("Error loading state, falling back to default:", e);
                    return DEFAULT_STATE;
                }
            }

            static save(state) {
                try {
                    localStorage.setItem('vyra_state_v1', JSON.stringify(state));
                } catch (e) {
                    console.error("Error saving state:", e);
                    showToast("Error saving data to local storage", "error");
                }
            }

            static reset() {
                localStorage.removeItem('vyra_state_v1');
                return DEFAULT_STATE;
            }
        }

        let appState = StorageManager.load();
        
        function saveState() {
            StorageManager.save(appState);

        // Audio Helper using Web Audio API
        const soundEngine = {
            playBeep() {
                if (!appState.timerSettings.soundEnabled) return;
                try {
                    const ctx = new (window.AudioContext || window.webkitAudioContext)();
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(880, ctx.currentTime); // A5
                    gain.gain.setValueAtTime(0.1, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.5);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.5);
                } catch(e) {
                    // Audio context blocked or not supported
                }
            }
        }

        class Router {
            constructor() {
                this.currentRoute = 'home';
                window.addEventListener('hashchange', () => this.handleRoute());
            }

            init() {
                const hash = window.location.hash.replace('#', '');
                this.navigate(hash || 'home', false);
            }

            navigate(route, updateHash = true) {
                this.currentRoute = route;
                if (updateHash) {
                    window.location.hash = route;
                }
                this.renderNav();
                this.renderView();
                window.scrollTo(0, 0);
            }

        function updateActiveWorkoutBanner() {
            const banner = document.getElementById('active-workout-banner');

            if (appState.activeSession) {

                const elapsedSec = Math.floor((Date.now() - appState.activeSession.startTime) / 1000);
                const mins = String(Math.floor(elapsedSec / 60)).padStart(2, '0');
                const secs = String(elapsedSec % 60).padStart(2, '0');
                document.getElementById('banner-timer').textContent = `${mins}:${secs}`;

        function toggleWidget(id) {
            const w = appState.dashboardWidgets.find(item => item.id === id);
            if (w) {
                w.enabled = !w.enabled;
                saveState();
                initHomeWidgets();
            }
        }

        function moveWidget(index, direction) {
            const newIndex = index + direction;
            if (newIndex < 0 || newIndex >= appState.dashboardWidgets.length) return;
            const temp = appState.dashboardWidgets[index];
            appState.dashboardWidgets[index] = appState.dashboardWidgets[newIndex];
            appState.dashboardWidgets[newIndex] = temp;
            // Re-assign orders
            appState.dashboardWidgets.forEach((w, i) => w.order = i + 1);
            saveState();
            openCustomizeDashboardModal();
            initHomeWidgets();
        }

        function initActiveWorkout() {
            if (activeTimerInterval) clearInterval(activeTimerInterval);
            activeTimerInterval = setInterval(() => {
                if (!appState.activeSession) return;
                const elapsedSec = Math.floor((Date.now() - appState.activeSession.startTime) / 1000);
                const mins = String(Math.floor(elapsedSec / 60)).padStart(2, '0');
                const secs = String(elapsedSec % 60).padStart(2, '0');
                const el = document.getElementById('workout-elapsed-time');
                if (el) el.textContent = `${mins}:${secs}`;
            }, 1000);
        }

        function updateSetActive(exIndex, setIndex, field, value) {
            if (!appState.activeSession) return;
            appState.activeSession.exercises[exIndex].sets[setIndex][field] = parseFloat(value) || 0;
            saveState();
        }

        function toggleSetComplete(exIndex, setIndex) {
            if (!appState.activeSession) return;
            const set = appState.activeSession.exercises[exIndex].sets[setIndex];
            set.completed = !set.completed;
            saveState();
            router.renderView();

            if (set.completed && appState.timerSettings.autoStart) {
                triggerSmartRestTimer(90, appState.activeSession.exercises[exIndex].name);
            }
        }

        function triggerSmartRestTimer(durationSeconds, exerciseName) {
            restTimerSecondsRemaining = durationSeconds;
            restTimerTotal = durationSeconds;
            document.getElementById('rest-exercise-title').textContent = exerciseName;
            document.getElementById('rest-timer-overlay').classList.remove('hidden');

            if (restTimerInterval) clearInterval(restTimerInterval);
            restTimerInterval = setInterval(() => {
                restTimerSecondsRemaining--;
                const mins = String(Math.floor(restTimerSecondsRemaining / 60)).padStart(2, '0');
                const secs = String(restTimerSecondsRemaining % 60).padStart(2, '0');
                document.getElementById('rest-time-display').textContent = `${mins}:${secs}`;
                
                const pct = (restTimerSecondsRemaining / restTimerTotal) * 100;
                document.getElementById('rest-progress-bar').style.width = `${pct}%`;

                if (restTimerSecondsRemaining <= 0) {
                    clearInterval(restTimerInterval);
                    soundEngine.playBeep();
                    dismissRestTimer();
                    showToast("Rest completed! Ready for next set.");
                }
            }, 1000);
        }

        function dismissRestTimer() {
            if (restTimerInterval) clearInterval(restTimerInterval);
            document.getElementById('rest-timer-overlay').classList.add('hidden');
        }

        function adjustRestTime(seconds) {
            restTimerSecondsRemaining = Math.max(0, restTimerSecondsRemaining + seconds);
            const mins = String(Math.floor(restTimerSecondsRemaining / 60)).padStart(2, '0');
            const secs = String(restTimerSecondsRemaining % 60).padStart(2, '0');
            document.getElementById('rest-time-display').textContent = `${mins}:${secs}`;
        }

        function finishActiveWorkout() {
            if (!appState.activeSession) return;
            if (activeTimerInterval) clearInterval(activeTimerInterval);

            const session = appState.activeSession;
            const durationMin = Math.round((Date.now() - session.startTime) / 60000) || 45;
            
            let totalSets = 0;
            let totalVolume = 0;
            session.exercises.forEach(ex => {
                ex.sets.forEach(s => {
                    if (s.completed) {
                        totalSets++;
                        totalVolume += (s.weight * s.reps);
                    }
                });
            });

            const completedWorkout = {
                id: 'w_' + Date.now(),
                routineId: session.routineId,
                routineName: session.routineName,
                date: new Date().toISOString(),
                durationMinutes: durationMin,
                exercisesCount: session.exercises.length,
                totalSets,
                totalVolume,
                prs: totalVolume > 5000 ? ['New Volume Milestone Achieved!'] : []
            };

            appState.workouts.unshift(completedWorkout);
            appState.activeSession = null;
            saveState();

            openModal(`
                <div class="text-center space-y-4">
                    <div class="w-16 h-16 rounded-3xl bg-emerald-500/10 text-emerald-500 flex items-center justify-center mx-auto text-2xl font-bold">✓</div>
                    <h3 class="text-2xl font-bold">Workout Complete!</h3>
                    <p class="text-sm text-light-muted dark:text-dark-muted">${completedWorkout.routineName}</p>
                    <div class="grid grid-cols-3 gap-2 py-4 border-y border-light-border dark:border-dark-border">
                        <div>
                            <p class="text-xs text-light-muted">Duration</p>
                            <p class="font-bold text-lg">${completedWorkout.durationMinutes}m</p>
                        </div>
                        <div>
                            <p class="text-xs text-light-muted">Sets</p>
                            <p class="font-bold text-lg">${completedWorkout.totalSets}</p>
                        </div>
                        <div>
                            <p class="text-xs text-light-muted">Volume</p>
                            <p class="font-bold text-lg">${completedWorkout.totalVolume} kg</p>
                        </div>
                    </div>
                    <button onclick="closeModal(); router.navigate('home');" class="w-full py-3.5 bg-brand-600 text-white font-semibold rounded-2xl shadow-lg">Save & Return Home</button>
                </div>
            `);
        }
    </script>
